cmake_minimum_required(VERSION 3.20)
project(navcraft_safety)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(cmd_vel_safety_gate_node src/cmd_vel_safety_gate_node.cpp)
ament_target_dependencies(cmd_vel_safety_gate_node rclcpp geometry_msgs std_msgs std_srvs)
install(TARGETS cmd_vel_safety_gate_node DESTINATION lib/${PROJECT_NAME})
ament_package()
