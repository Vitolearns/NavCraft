cmake_minimum_required(VERSION 3.20)
project(navcraft_simulation)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(planar_odometry src/planar_odometry.cpp)
target_include_directories(planar_odometry PUBLIC include)
ament_target_dependencies(planar_odometry nav_msgs tf2)
add_executable(gazebo_odom_tf_node src/gazebo_odom_tf_node.cpp)
target_link_libraries(gazebo_odom_tf_node planar_odometry)
ament_target_dependencies(gazebo_odom_tf_node rclcpp geometry_msgs nav_msgs tf2_ros)
install(TARGETS planar_odometry DESTINATION lib)
install(TARGETS gazebo_odom_tf_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch rviz worlds media DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(planar_odometry_test test/planar_odometry_test.cpp)
  target_link_libraries(planar_odometry_test planar_odometry)
  ament_target_dependencies(planar_odometry_test nav_msgs tf2)
endif()
ament_package()
